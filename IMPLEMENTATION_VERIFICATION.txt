━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Install ONNX export dependencies:
   $ pip install xgboost skl2onnx onnx onnxmltools scikit-learn

2. Train your multi-output models in Python

3. Export models to ONNX:
   $ python3 export_models_to_onnx.py --model-type xgboost ...

4. Build C++ application:
   $ cd cpp_inference/build
   $ cmake .. -DCMAKE_BUILD_TYPE=Release
   $ cmake --build .

5. Run inference:
   $ ./radar_tagger_multioutput --model model.onnx --model-type xgboost
